================================================================
 EC7207 - HPC Network Traffic Analysis
 Complete Compilation & Run Guide
================================================================

IMPORTANT: Always run Serial first — all other implementations
           read serial_time.txt to calculate speedup & efficiency.

DATA PATH: data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv
           (use this full path in all commands below)

================================================================
 STEP 1 — CREATE RESULTS DIRECTORY
================================================================

    mkdir -p results/logs


================================================================
 STEP 2 — SERIAL (run this first, always)
================================================================

    C1. Compile serial baseline:
   gcc -O3 -march=native -o serial src/serial/network_analysis_serial.c -lm

2. Run serial baseline:
   ./serial data/UNSW-NB15_1.csv/UNSW-NB15_1_with_header.csv

3. Output results:
   - Confusion matrix
   - Accuracy, Precision, Recall, F1
   - Execution time (single-pass + total)
   - Saved logs in results/logs/



================================================================
 STEP 3 — OpenMP (shared memory, multiple threads)
================================================================

    Compile:
        gcc -Wall -O2 -std=c11 -fopenmp -lm \
            -o results/openmp \
            src/openmp/network_analysis_openmp.c

    Run (1, 2, 4, 8 threads):
        OMP_NUM_THREADS=1 ./results/openmp \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        OMP_NUM_THREADS=2 ./results/openmp \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        OMP_NUM_THREADS=4 ./results/openmp \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        OMP_NUM_THREADS=8 ./results/openmp \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv


================================================================
 STEP 4 — POSIX Threads (shared memory, manual threads)
================================================================

    Compile:
        gcc -Wall -O2 -std=c11 -pthread -lm \
            -o results/pthreads \
            src/pthreads/network_analysis_pthread.c

    Run (1, 2, 4, 8 threads — thread count is 2nd argument):
        ./results/pthreads \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 1

        ./results/pthreads \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 2

        ./results/pthreads \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 4

        ./results/pthreads \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 8


================================================================
 STEP 5 — MPI (distributed memory, multiple processes)
================================================================

    Compile:
        mpicc -Wall -O2 -std=c11 -lm \
            -o results/mpi \
            src/mpi/network_analysis_mpi.c

    Run (1, 2, 4, 8 processes):
        mpirun -np 1 ./results/mpi \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        mpirun -np 2 ./results/mpi \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        mpirun -np 4 ./results/mpi \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        mpirun -np 8 ./results/mpi \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv


================================================================
 STEP 6 — HYBRID MPI + OpenMP (two-level parallelism)
================================================================

    Compile:
        mpicc -Wall -O2 -std=c11 -fopenmp -lm \
            -o results/hybrid \
            src/hybrid/network_analysis_hybrid.c

    Run (set OMP_NUM_THREADS before mpirun):

        2 ranks x 2 threads = 4-way:
        OMP_NUM_THREADS=2 mpirun -np 2 ./results/hybrid \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        2 ranks x 4 threads = 8-way:
        OMP_NUM_THREADS=4 mpirun -np 2 ./results/hybrid \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        4 ranks x 2 threads = 8-way:
        OMP_NUM_THREADS=2 mpirun -np 4 ./results/hybrid \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

        4 ranks x 4 threads = 16-way:
        OMP_NUM_THREADS=4 mpirun -np 4 ./results/hybrid \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv


================================================================
 STEP 7 — CUDA (GPU acceleration)
================================================================

    Compile:
        nvcc -O2 -std=c++11 \
            -o results/cuda \
            src/cuda/network_analysis_cuda.cu

    Run (default block size = 256):
        ./results/cuda \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

    Run with custom block size:
        ./results/cuda \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 128

        ./results/cuda \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 256

        ./results/cuda \
            data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv 512

    Check if CUDA/GPU is available:
        nvidia-smi
        nvcc --version


================================================================
 QUICK REFERENCE — ALL COMPILE COMMANDS
================================================================

    gcc -Wall -O2 -std=c11 -lm          -o results/serial   src/serial/network_analysis_serial.c
    gcc -Wall -O2 -std=c11 -fopenmp -lm -o results/openmp   src/openmp/network_analysis_openmp.c
    gcc -Wall -O2 -std=c11 -pthread -lm -o results/pthreads src/pthreads/network_analysis_pthread.c
    mpicc -Wall -O2 -std=c11 -lm        -o results/mpi      src/mpi/network_analysis_mpi.c
    mpicc -Wall -O2 -std=c11 -fopenmp -lm -o results/hybrid src/hybrid/network_analysis_hybrid.c
    nvcc -O2 -std=c++11                 -o results/cuda     src/cuda/network_analysis_cuda.cu


================================================================
 QUICK REFERENCE — RECOMMENDED RUN ORDER
================================================================

    DATA=data/UNSW_NB15_training-set.csv/UNSW_NB15_training-set.csv

    ./results/serial $DATA                            # STEP 1 - always first

    OMP_NUM_THREADS=4 ./results/openmp   $DATA        # STEP 2
    ./results/pthreads $DATA 4                        # STEP 3
    mpirun -np 4 ./results/mpi $DATA                  # STEP 4
    OMP_NUM_THREADS=4 mpirun -np 2 ./results/hybrid $DATA   # STEP 5
    ./results/cuda $DATA                              # STEP 6

================================================================
 TROUBLESHOOTING
================================================================

    "cannot open source file mpi.h" in VS Code:
        → Not a real error. Compile from terminal, it will work.
        → To fix squiggles: add /usr/lib/x86_64-linux-gnu/openmpi/include
          to .vscode/c_cpp_properties.json includePath.
        → Find mpi.h location: find /usr -name "mpi.h" 2>/dev/null

    "No CUDA GPU found":
        → Run: nvidia-smi  (check if GPU is detected)
        → CUDA code requires an NVIDIA GPU with CUDA drivers installed.

    Speedup not showing:
        → Make sure serial was run first and serial_time.txt exists.
        → Check: cat results/serial_time.txt  or  cat serial_time.txt

    mpirun permission error:
        → Try: mpirun --allow-run-as-root -np 4 ./results/mpi $DATA
